	\section{Introducción}
	
	\begin{frame}{Problema y Motivación}
		\justifying
		La inspección autónoma en entornos complejos (ej. minería a cielo abierto) carece de rutas definidas, requiriendo que los robots construyan mapas y decidan su próximo movimiento de forma inteligente.
		
		\vspace{1ex}
		\textbf{Limitaciones Actuales:}
		\begin{itemize}
			\item \textbf{Desacoplamiento:} Los planificadores tradicionales buscan ganancia geométrica pero son "ciegos" al contexto semántico.
			\item \textbf{Costo Computacional:} Dependencia de procesamiento en la nube (Cloud) para tareas complejas, impidiendo el uso en drones (UAVs) en tiempo real.
			\item \textbf{Dependencia Humana:} Se requiere etiquetado manual para identificar áreas de interés.
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Objetivo del Survey}
		\justifying
		Proponer y analizar la viabilidad de una \textbf{arquitectura de ciclo cerrado} unificando percepción y cognición en hardware Edge.
		
		\vspace{1ex}
		\textbf{Pilares Tecnológicos:}
		\begin{itemize}
			\item \textbf{3DGS (3D Gaussian Splatting):} Reconstrucción fotorealista explícita y rápida en tiempo real.
			\item \textbf{VLA (Vision-Language-Action):} Modelos de inteligencia semántica y ejecución de comandos.
			\item \textbf{Pred-NBV (Predictive Next-Best-View):} Planificación predictiva para maximizar cobertura y minimizar esfuerzo de vuelo.
		\end{itemize}
	\end{frame}

	\begin{frame}{Modulación de Incertidumbre Semántica}
		\justifying
		A diferencia de los enfoques geométricos tradicionales, la arquitectura usa el VLA para priorizar áreas críticas (ej. grietas en minería).
		
		\vspace{1ex}
		\textbf{Campo de Incertidumbre Ponderado por Semántica ($U_s$):}
		\begin{equation}
			U_s(x) = \int_{p \in P} w_{\text{vla}}(p)\,\phi(g,r)\, dp
		\end{equation}
		
		\vspace{1ex}
		\begin{itemize}
			\item $w_{\text{vla}}(p)$: Peso probabilístico dado por el VLA según la relevancia del objetivo.
			\item Prioriza el re-escaneo de elementos de interés estructural.
			\item Ignora activamente el fondo (alta entropía irrelevante como cielo o vegetación).
		\end{itemize}
	\end{frame}

	\begin{frame}{Factibilidad en Hardware Edge (NVIDIA Jetson)}
		\justifying
		El principal reto de computación a bordo se conoce como \textit{"Memory Wall"}.
		Se resuelve usando adaptación de bajo rango (\textbf{LoRA}) y cuantización (\textbf{INT8}).
		
		\vspace{2ex}
		\begin{table}[htbp]
			\caption{Comparación Teórica de Consumo VRAM y Latencia}
			\centering
			\resizebox{0.8\textwidth}{!}{
				\begin{tabular}{l l c c}
					\toprule
					\textbf{Arquitectura VLA} & \textbf{Método de Optimización} & \textbf{Latencia (s)} & \textbf{VRAM (GB)} \\
					\midrule
					\textbf{Nube (Cloud)} & Full Fine-Tuning (Tradicional) & $> 2.50$ & $\approx 40 - 80$ \\

			\item \textbf{LoRA:} Cero latencia adicional en inferencia al fusionarse los pesos linealmente.
			\item Memoria Unificada elimina transferencias PCIe, ejecutando VLA y 3DGS simultáneamente.
		\end{itemize}
	\end{frame}

	\section{Discusión y Retos Abiertos}
	
	\begin{frame}{Desafíos en Entornos Dinámicos Reales}
		\begin{columns}
			\begin{column}{0.5\textwidth}
				\textbf{Dinámica de Escena y 4D-GS}
				\begin{itemize}
					\item \textbf{Problema:} Maquinaria pesada móvil arruina el mapa estático 3DGS.
					\item \textbf{Solución:} Evolución a representaciones 4D-GS. Permite aislar temporalmente la infraestructura estática de los obstáculos dinámicos.
				\end{itemize}
			\end{column}
			\begin{column}{0.5\textwidth}
				\textbf{Brecha "Sim-to-Real" y Datos}
				\begin{itemize}
					\item \textbf{Problema:} Políticas entrenadas en simulación fallan por iluminación extrema o polvo (minería).
					\item \textbf{Solución:} Adaptación de dominio adversarial y \textit{fine-tuning} local rápido in situ.
				\end{itemize}
			\end{column}
		\end{columns}
		
		\vspace{2ex}
		\textbf{Certificación de Seguridad (Safety-Critical):} Los resultados de redes neuronales (VLA) son probabilísticos. Urge usar \textit{Control Barrier Functions (CBF)} para validar trayectorias antes de ser ejecutadas.
	\end{frame}
	
	% =======================================================
	\section{Conclusiones}
	
	\begin{frame}{Conclusiones}
		\justifying
		\begin{itemize}
			\item \textbf{Viabilidad Edge:} Integrar VLA cuantizados a INT8 con optimizaciones LoRA y reconstrucción explícita (3DGS) es computacionalmente factible en ecosistemas de memoria unificada ($\approx 8-12$ GB, latencia $< 1$s).
			\item \textbf{Cambio de Paradigma:} El \textit{Semantic-Metric Loop} cambia el enfoque de un mapeo geométrico pasivo a una exploración completamente autónoma guiada por intención.
			\item \textbf{Próximos Pasos Industriales:} Madurar este sistema hacia activos industriales fiables (como en minería) exige obligatoriamente marcos de certificación de seguridad, modelado 4D dinámico y coordinación de enjambres bajo restricciones de red.
		\end{itemize}
	\end{frame}
